# 4축 진단 체계 재편 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 진단 축을 3개(`legal`/`physical`/`regulatory`)에서 4개(`rights`/`tax`/`property`/`market`)로 재편하고, 해결형 판정 `선행필요`와 재산세 자동 계산을 추가한다.

**Architecture:** 축은 4개지만 **등급은 `rights`·`property` 두 축만 결정**한다(`GATING`). `tax`·`market`은 축 판정만 내고 서술과 경로 정렬에 쓰인다. 새 등급 `선행필요`(verdict `precondition`)는 "절차를 밟으면 풀린다"를 뜻하며 **사람이 등기를 확인했을 때만** 찍힌다 — 공개 API로는 미등기를 판별할 수 없기 때문이다. 재산세는 순수 함수로 계산하고 세율·비율은 `tax.json`에 데이터로 둔다.

**Tech Stack:** Next.js 15 App Router · React 19 · TypeScript 5.7 · Tailwind v4 · Supabase REST(SDK 없음) · tsx · **node:test**(Node v25 내장, 신규 의존성 없음)

**Spec:** [`docs/superpowers/specs/2026-09-06-4축-진단체계-design.md`](../specs/2026-09-06-4축-진단체계-design.md)

- [ ] **Step 5: `src/tax.ts` 작성**

```ts
import taxJson from './rules/tax.json' with { type: 'json' }

/** 누진 구간. base 는 구간 시작점까지의 누적 세액, rate 는 초과분에 붙는 요율. */
type Bracket = { upto: number | null; base: number; rate: number }

type TaxTable = {
  version: string
  singleHomeCeiling: number
  fairMarketRatio: { single: number; general: number }
  rates: { single: Bracket[]; general: Bracket[] }
  localEduTaxRate: number
  urbanAreaRate: number
  urbanZones: string[]
}

const T = taxJson as TaxTable

export type PropertyTax = {
  /** 1세대1주택 특례 적용값 (공시 9억 이하) */
  single: number
  /** 표준세율 적용값 */
  general: number
  /** 도시지역분이 포함됐는지 */
  urbanIncluded: boolean
  /** 어느 해 기준표를 썼는지 */
  year: string
}

function bracketTax(taxBase: number, brackets: Bracket[]): number {
  let floorOfBracket = 0
  for (const b of brackets) {
    if (b.upto === null || taxBase <= b.upto) {
      return Math.floor(b.base + (taxBase - floorOfBracket) * b.rate)
    }
    floorOfBracket = b.upto
  }
  return 0
}

/**
 * 주택 재산세 추정.
 *
 * 소유주의 주택 수를 우리는 모른다 (폼에서 묻지 않는다).
 * 그래서 하나의 값이 아니라 1세대1주택과 표준세율 두 값을 돌려주고,
 * 진단서가 "1주택이시면 X, 다른 집이 있으시면 Y"로 서술한다.
 */
export function propertyTax(
  housePrice: number | null | undefined,
  zone1: string | null | undefined,
): PropertyTax | null {
  const price = Number(housePrice)
  if (!Number.isFinite(price) || price <= 0) return null

  // 도시지역분은 국토계획법상 도시지역에만 붙는다.
  // 관리·농림·자연환경보전지역인 시골 필지에는 부과되지 않는다.
  const zone = String(zone1 ?? '')
  const urbanIncluded = T.urbanZones.some((z) => zone.includes(z))

  // 공시 9억을 넘으면 1세대1주택 특례를 받지 못한다.
  const eligibleForSingle = price <= T.singleHomeCeiling

  const compute = (kind: 'single' | 'general'): number => {
    const taxBase = Math.floor(price * T.fairMarketRatio[kind])
    const main = bracketTax(taxBase, T.rates[kind])
    const localEdu = Math.floor(main * T.localEduTaxRate)
    const urban = urbanIncluded ? Math.floor(taxBase * T.urbanAreaRate) : 0
    return main + localEdu + urban
  }

  const general = compute('general')

  return {
    single: eligibleForSingle ? compute('single') : general,
    general,
    urbanIncluded,
    year: T.version,
  }
}

const man = (n: number): string => (n / 10_000).toFixed(1) + '만'

/** "연 약 1.6만 ~ 4.4만원" */
export function formatTaxRange(t: PropertyTax): string {
  return `연 약 ${man(t.single)} ~ ${man(t.general)}원`
}
```

```ts
import { combine, summarize, AXIS_ORDER } from './verdict'

const f = (axis: Finding['axis'], verdict: Finding['verdict'], label = 'L'): Finding => ({

test('rights 가 precondition 이면 선행필요', () => {
```

```ts
import type { Finding } from './rules/engine'
import type { AxisVerdict } from './types'

/** 심각도. 축별로 가장 심한 것이 그 축의 판정이 된다. */
const RANK: Record<AxisVerdict, number> = {
  clear: 0, unknown: 1, suspect: 2, precondition: 3, blocked: 4,
}

export type Axis = 'rights' | 'tax' | 'property' | 'market'

/** 진단서 표시 순서. 4개 전부 나온다. */
export const AXIS_ORDER: Axis[] = ['rights', 'tax', 'property', 'market']

/**
 * 등급을 결정하는 축.
 *
 * 세금이 나쁘다고 거래가 막히지는 않는다 — 셈법이 바뀔 뿐이다. 시장성도 같다.
 * rights·property 는 '할 수 있느냐'를 막고, tax·market 은 '어느 쪽이 유리하냐'를 정한다.
 * tax·market 도 축 판정(verdict)은 내되 여기에는 들어오지 않는다.
 */
const GATING: Axis[] = ['rights', 'property']

export const AXIS_LABEL: Record<Axis, string> = {
  rights: '① 권리',
  tax: '② 세금',
  property: '③ 건물·토지',
  market: '④ 시장·관리',
}

export type AxisSummary = {
  axis: Axis
  verdict: AxisVerdict
  findings: Finding[]
}

export type Grade = '가능' | '조건부' | '선행필요' | '불가'

export type Diagnosis = {
  grade: Grade
  /** 등급을 결정한 축 (가능이면 null) */
  decidedBy: Axis | null
  headline: string
  axes: AxisSummary[]
  /** 사람이 현장 확인을 마쳤는지. 자동 판정만으로는 '가능'이 나오지 않는다 */
  fieldVerified: boolean
}

export function summarize(findings: Finding[]): AxisSummary[] {
  return AXIS_ORDER.map((axis) => {
    const own = findings.filter((f) => f.axis === axis)
    const verdict = own.reduce<AxisVerdict>(
      (worst, f) => (RANK[f.verdict] > RANK[worst] ? f.verdict : worst),
      'clear',
    )
    // 같은 항목(label)에 문제가 잡혔으면 그 항목의 clear 는 지운다 — 상충 표시 방지.
    // 다른 항목의 clear 는 남긴다. '위반건축물 등재 없음' 같은 확인 사실은
    // 축 전체가 unknown 이라는 이유로 사라지면 안 된다.
    const troubled = new Set(own.filter((f) => f.verdict !== 'clear').map((f) => f.label))
    const findingsOut = own.filter((f) => f.verdict !== 'clear' || !troubled.has(f.label))

    // 심각한 것부터 보여준다
    findingsOut.sort((a, b) => RANK[b.verdict] - RANK[a.verdict])

    return { axis, verdict, findings: findingsOut }
  })
}

export function combine(findings: Finding[], fieldVerified = false): Diagnosis {
  const axes = summarize(findings)

  // 등급은 GATING 축만 본다. 우선순위는 앞선 축이 이긴다.
  const gating = GATING.map((a) => axes.find((x) => x.axis === a)!)
  const worstOf = (v: AxisVerdict) => gating.find((x) => x.verdict === v)

  const blocked = worstOf('blocked')
  if (blocked) {
    return {
      grade: '불가',
      decidedBy: blocked.axis,
      headline: blocked.findings[0]?.reason ?? '거래를 막는 사유가 있습니다.',
      axes,
      fieldVerified,
    }
  }

  // 해결형 — "불가"가 아니라 "이것부터 하면 됩니다"
  const pre = worstOf('precondition')
  if (pre) {
    return {
      grade: '선행필요',
      decidedBy: pre.axis,
      headline: pre.findings[0]?.reason ?? '먼저 정리해야 할 절차가 있습니다.',
      axes,
      fieldVerified,
    }
  }

  // 구체적 문제(suspect)가 있으면 그것을 헤드라인으로 쓴다
  const concern = worstOf('suspect')
  if (concern) {
    return {
      grade: '조건부',
      decidedBy: concern.axis,
      headline: concern.findings[0]?.reason ?? '확인이 더 필요합니다.',
      axes,
      fieldVerified,
    }
  }

  // 남은 게 '자동으로 확인 못 한 것'뿐이라면, 그 사실 자체가 좋은 소식이다.
  // 모든 진단서가 "등기부는 공개 API가 없어…" 로 시작하면 아무 정보가 안 된다.
  const unknown = worstOf('unknown')
  if (unknown) {
    return {
      grade: '조건부',
      decidedBy: unknown.axis,
      headline:
        '공개 자료에서는 거래를 막는 사유가 확인되지 않았습니다. 등기와 현장 확인이 남았습니다.',
      axes,
      fieldVerified,
    }
  }

  // 두 축이 모두 clear 여도 현장 확인 전에는 '가능'을 주지 않는다
  if (!fieldVerified) {
    return {
      grade: '조건부',
      decidedBy: null,
      headline: '공개 자료상으로는 막는 사유가 없습니다. 현장 확인이 남았습니다.',
      axes,
      fieldVerified,
    }
  }

  return {
    grade: '가능',
    decidedBy: null,
    headline: '지금 처분할 수 있는 상태입니다.',
    axes,
    fieldVerified,
  }
}
```

```ts
import type { Finding } from './rules/engine'

/**
 * 등기 확인 결과. 공개 API 가 없어 사람이 등기소에서 보고 적는다.
 *
 * 각 칸은 3상태다: true / false / null(아직 안 봄).
 * null 을 false 로 뭉개면 "확인해보니 없더라"와 "아직 안 봤다"가 같아진다.
 */
export type Registry = {
  /** 건물등기부가 있는가 */
  buildingRegistered: boolean | null
  /** 토지 소유자와 건물 소유자가 같은가 */
  sameOwner: boolean | null
  /** 지상권·임차권이 설정돼 있는가 */
  superficies: boolean | null
  /** 근저당·가압류가 있는가 */
  encumbrance: boolean | null
  /** 운영자가 손으로 덧붙이는 말 */
  note: string | null
  /** 등기소에서 확인한 날 YYYY-MM-DD */
  checkedAt: string | null
}

export const EMPTY_REGISTRY: Registry = {
  buildingRegistered: null, sameOwner: null, superficies: null,
  encumbrance: null, note: null, checkedAt: null,
}

/**
 * §2.1 의 아래쪽 자물쇠.
 * 4칸이 모두 채워졌을 때만 참이고, 그때만 precondition 이 나올 수 있다.
 */
export function registryVerified(r: Registry): boolean {
  return (
    r.buildingRegistered !== null &&
    r.sameOwner !== null &&
    r.superficies !== null &&
    r.encumbrance !== null
  )
}

const SOURCE = '등기사항증명서'

const make = (
  ruleId: string,
  verdict: Finding['verdict'],
  label: string,
  reason: string,
  nextStep?: string,
): Finding => ({ ruleId, axis: 'rights', verdict, label, reason, nextStep, source: SOURCE })

/**
 * 등기 4칸을 Finding 으로 바꾼다.
 *
 * precondition 은 rules.json 이 만들지 않는다 — rules.json 은 공개 데이터만 다루고
 * 등기는 공개 데이터가 아니다. 그래서 여기서 직접 만든다.
 */
export function registryFindings(r: Registry): Finding[] {
  if (!registryVerified(r)) {
    return [
      make(
        'REG-UNKNOWN', 'unknown', '등기',
        '등기부는 공개 API가 없어 자동으로 확인할 수 없습니다.',
        '등기사항전부증명서를 열람해 상속 미등기·근저당·가압류를 확인해야 합니다.',
      ),
    ]
  }

  const out: Finding[] = []

  if (r.buildingRegistered === false) {
    out.push(make(
      'REG-NO-BUILDING', 'precondition', '건물등기',
      '건물등기부가 없습니다. 이 상태로는 소유권을 넘길 수 없어 매각이 진행되지 않습니다.',
      '소유권보존등기를 먼저 하셔야 합니다. 건축물대장이 있으면 법무사를 통해 진행할 수 있습니다.',
    ))
  }

  // 소유자가 다르다는 것만으로는 선행 절차가 아니다.
  // 지상권이 있으면 건물은 지켜지고 매각도 된다 — 매수인 설명 부담만 남는다.
  if (r.sameOwner === false) {
    out.push(
      r.superficies
        ? make(
            'REG-SPLIT-WITH-RIGHT', 'suspect', '토지사용 권원',
            '토지 소유자와 건물 소유자가 다르지만 지상권·임차권이 설정돼 있습니다. 건물은 유지됩니다.',
            '지료와 존속기간을 확인하고, 매각 시 매수인에게 설명해야 합니다.',
          )
        : make(
            'REG-SPLIT-NO-RIGHT', 'precondition', '토지사용 권원',
            '토지 소유자와 건물 소유자가 다른데 지상권·임차권이 확인되지 않습니다.',
            '법정지상권(민법 §366·관습법상) 성립 여부를 법무사와 확인하시거나, 토지 소유자와 사용 권원을 정리하셔야 합니다.',
          ),
    )
  }

  if (r.encumbrance === true) {
    out.push(make(
      'REG-ENCUMBRANCE', 'suspect', '근저당·가압류',
      '근저당 또는 가압류가 설정돼 있습니다.',
      '매각 시 말소 절차와 잔금 처리 순서를 미리 정해두셔야 합니다.',
    ))
  }

  if (out.length === 0) {
    out.push(make(
      'REG-CLEAR', 'clear', '등기',
      '건물등기·소유자·지상권·근저당에서 걸리는 것이 확인되지 않았습니다.',
    ))
  }

  return out
}

/** 새 jsonb 를 읽되, 없으면 옛 registry_note 로 폴백한다. */
export function parseRegistry(row: {
  registry?: unknown
  registry_note?: string | null
  registry_checked_at?: string | null
}): Registry {
  const r = row.registry
  if (r && typeof r === 'object' && !Array.isArray(r)) {
    return { ...EMPTY_REGISTRY, ...(r as Partial<Registry>) }
  }
  // 옛 row — 4칸은 채울 수 없다. note 와 날짜만 살린다.
  return {
    ...EMPTY_REGISTRY,
    note: row.registry_note ?? null,
    checkedAt: row.registry_checked_at ?? null,
  }
}
```

```ts
export type Item = {
  label: string
  lines: string[]
  source: string | null
  /** 확인하지 못한 항목. 화면에서 배경과 바로 구분한다 */
  unverified: boolean
}

export type AxisBlock = {
  axis: Axis
  /** '① 권리' */
  label: string
  /** 축 헤더에 붙는 한국어 배지 */
  badge: string
  items: Item[]
}

const BADGE: Record<string, string> = {
  clear: '양호',
  unknown: '확인 필요',
  suspect: '조건부',
  precondition: '선행 필요',
  blocked: '불가',
}

const MISSING = '확인하지 못했습니다.'

/** 값이 있으면 실선, 없으면 미확인. 같은 판단을 열 번 쓰지 않는다. */
const item = (label: string, lines: string[], source: string | null): Item =>
  lines.length > 0
    ? { label, lines, source, unverified: false }
    : { label, lines: [MISSING], source: null, unverified: true }

export function buildAxisBlocks(
  axes: Axes | null,
  facts: Record<string, unknown>,
  note: string | null,
  registry: Registry,
): AxisBlock[] {
  const checked = axes?.checkedAt ? dot(axes.checkedAt) : null
  const src = (name: string) => (checked ? `${name} ${checked} 확인` : name)
  const verdictOf = (axis: Axis) =>
    axes?.diagnosis?.axes?.find((a) => a.axis === axis)?.verdict ?? 'unknown'

  // ── ① 권리 ──────────────────────────────
  const regDone = registryVerified(registry)
  const legacyNote = Boolean(!regDone && registry.note && registry.checkedAt)
  const registryLines = regDone
    ? [
        registry.buildingRegistered ? '건물등기부 있음' : '건물등기부 없음 (미등기)',
        registry.sameOwner ? '토지·건물 소유자 같음' : '토지·건물 소유자 다름',
        registry.superficies ? '지상권·임차권 있음' : '지상권·임차권 없음',
        registry.encumbrance ? '근저당·가압류 있음' : '근저당·가압류 없음',
        ...(registry.note ? registry.note.split('\n').filter(Boolean) : []),
      ]
    : legacyNote
      ? registry.note!.split('\n').filter(Boolean)
      : []

  const ownerLines = [
    facts.ownerType ? `토지 소유구분 ${facts.ownerType}` : null,
    typeof facts.coOwnerCount === 'number' && facts.coOwnerCount >= 2
      ? `공유인 ${facts.coOwnerCount}명`
      : null,
    facts.ownershipCause ? `소유권 변동 원인 ${facts.ownershipCause}` : null,
  ].filter((s): s is string => s !== null)

  // ── ② 세금 ──────────────────────────────
  const tax = propertyTax(facts.housePrice as number, facts.zone1 as string)
  const holdingLines = tax
    ? [
        `${formatTaxRange(tax)} (추정)`,
        `1세대1주택이시면 ${(tax.single / 10_000).toFixed(1)}만원, 다른 집이 있으시면 ${(tax.general / 10_000).toFixed(1)}만원 수준입니다.`,
        `공시가격 ${won(facts.housePrice)} 기준 · 종합부동산세는 대상이 아닙니다.`,
      ]
    : []

  // 계산하지 않는다 — 세무사법 §2. 어느 쪽을 보셔야 하는지까지만 말한다.
  const disposalLines = [
    '다른 집이 없으시다면 — 1세대1주택 비과세 요건(2년 보유 등)을 보시면 됩니다.',
    '다른 집이 있으시다면 — 상속주택 특례(소득세법 시행령 §155②)와 농어촌주택 특례(조특법 §99의4) 두 가지를 확인해볼 수 있습니다.',
    '구체적인 세액은 이 진단서에서 계산하지 않습니다. 세무사 확인이 필요합니다.',
  ]

  // ── ③ 건물·토지 ─────────────────────────
  const building = [
    facts.useApprovalDate ? `사용승인 ${dot(facts.useApprovalDate)}` : null,
    [facts.mainPurpose, facts.structure, facts.floors && `지상 ${facts.floors}층`]
      .filter(Boolean).join(' · ') || null,
    facts.buildingArea ? `연면적 ${facts.buildingArea}㎡` : null,
  ].filter((s): s is string => Boolean(s))

  const road = findingsBy(axes, ['진입로', '지형'])
  const regulation = [facts.zone1, ...(Array.isArray(facts.zones) ? facts.zones.slice(1) : [])]
    .filter(Boolean).join(' · ')

  // ── ④ 시장·관리 ─────────────────────────
  const price = won(facts.housePrice)

  return [
    {
      axis: 'rights',
      label: AXIS_LABEL.rights,
      badge: BADGE[verdictOf('rights')] ?? '확인 필요',
      items: [
        {
          label: '등기',
          lines: registryLines.length ? registryLines : ['등기소에서 확인이 필요합니다.'],
          source: regDone || legacyNote ? `등기사항증명서 ${dot(registry.checkedAt)} 확인` : null,
          unverified: !regDone && !legacyNote,
        },
        item('토지 소유', ownerLines, ownerLines.length ? src('토지소유정보') : null),
      ],
    },
    {
      axis: 'tax',
      label: AXIS_LABEL.tax,
      badge: BADGE[verdictOf('tax')] ?? '참고',
      items: [
        item('보유세', holdingLines, tax ? `지방세법 §111·§111의2 · ${tax.year}년 기준 추정` : null),
        item('처분세', disposalLines, '소득세법 · 조세특례제한법'),
      ],
    },
    {
      axis: 'property',
      label: AXIS_LABEL.property,
      badge: BADGE[verdictOf('property')] ?? '확인 필요',
      items: [
        item('건축물', building, building.length ? src('건축물대장') : null),
        {
          label: '진입로',
          lines: road.length ? road.map((f) => f.reason ?? '') : [MISSING],
          source: road.length ? src('토지특성') : null,
          unverified: road.length === 0 || road.some((f) => f.verdict === 'unknown'),
```
